\Needspace{10\baselineskip}\section*{A. Color Charts and Sensitivity}\addcontentsline{toc}{section}{A. Color Charts and Sensitivity}\label{app:color-charts}

\textbf{Injective and Lipschitz color charts.} Let $\Phi:\Omega_{\mathrm{RGB}}\to\Omega_*\subset\mathbb R^3$ be any map that is injective on the finite $256^3$ sRGB grid without coordinate rounding or bin merging. Category relabeling preserves exact-code JSD ($D_{\mathrm{JSD}}(\Phi_\#P\|\Phi_\#Q)=D_{\mathrm{JSD}}(P\|Q)$) and Proposition 1's support-escape events, while strong negative type on bounded $\Omega_*$ \cite{ref26,ref30} ensures $\tilde D_{\mathrm{E},\delta_*}(\Phi_\#P,\Phi_\#Q)=0\iff P=Q$ ($\delta_*>0$). If $\|\Phi(x)-\Phi(x')\|_2\le L_{\max}\|x-x'\|_2$ ($L_{\max}<\infty$), an RGB100 shift of at most $\epsilon$ moves $\Phi$-coordinates by at most $L_{\max}\epsilon$. Proposition 2 then gives one- and two-sided bounds $2L_{\max}\epsilon/\delta_*$ and $4L_{\max}\epsilon/\delta_*$ (capped at $1$ for clipped ratios) for $D_*(U,V)=\tilde D_{\mathrm{E},\delta_*}(\Phi(U),\Phi(V))$.


\textbf{Uniformly scaled D65 CIELAB control.} Standard D65-relative sRGB-to-CIELAB conversion \cite{ref24,ref31,ref32} (piecewise linear/power transfer without D50 adaptation) is injective on the 8-bit grid, but its nonlinear perceptual warping shifts finite-sample energy rankings relative to undistorted RGB100 ($L_{\max}=1$). Table S1 applies a single frozen isotropic scale ($0.7658$, fitted from within-cell pairwise distances in B-20\ensuremath{\times}6 enrollment data alone), mapping $\delta_0=15$ to native CIELAB $\delta_*=19.59$ without channelwise scaling or clipping.

\begin{proof}[Proof of Proposition 1]

Let $q=Q^{\mathrm{jitter}}(\Omega\setminus\operatorname{supp}(P_{\mathrm{ref}}))\ge t$; when $\operatorname{supp}(P_{\mathrm{src}})\subseteq\{a,\ldots,255-a\}^3$, each source color produces $(2a+1)^3$ equiprobable outcomes of which at most $r$ lie in $\operatorname{supp}(P_{\mathrm{ref}})$, giving $q\ge\max(0,1-r/(2a+1)^3)$. Convexity of $Q\mapsto D_{\mathrm{JSD}}(P_{\mathrm{ref}}\|Q)$ and $\mathbb E[\hat Q_n^{\mathrm{jitter}}]=Q^{\mathrm{jitter}}$ yield $\mathbb E[\Dhex{}(P_{\mathrm{ref}},\hat Q_n^{\mathrm{jitter}})]\ge D_{\mathrm{JSD}}(P_{\mathrm{ref}}\|Q^{\mathrm{jitter}})$ by Jensen's inequality (in main-manuscript Fig. 3(a), conditioning on the observed clean query sample fixes $P_{\mathrm{src}}$ at the empirical query histogram and takes expectation over one independent jitter draw per observation). Coarsening $\Omega$ via $T(x)=\mathbf 1[x\notin\operatorname{supp}(P_{\mathrm{ref}})]$ maps $P_{\mathrm{ref}}$ to $(1,0)$ and $Q^{\mathrm{jitter}}$ to $(1-q,q)$; because $D_{\mathrm{JSD}}$ is a Csisz\'{a}r $f$-divergence with strictly convex generator \cite{ref33}, data processing gives $D_{\mathrm{JSD}}(P_{\mathrm{ref}}\|Q^{\mathrm{jitter}})\ge D_{\mathrm{JSD}}((1,0)\|(1-q,q))=\psi(q)$ with $\psi$ in main-manuscript Eq. (1). Since $\psi'(q)=\tfrac12\big(1-\log_2\tfrac{2(1-q)}{2-q}\big)>\tfrac12>0$ on $(0,1)$, $\psi(q)\ge\psi(t)\ge t/2$. Equality $D_{\mathrm{JSD}}(P_{\mathrm{ref}}\|Q^{\mathrm{jitter}})=\psi(t)$ holds iff $q=t$ and $Q^{\mathrm{jitter}}(x)=(1-t)P_{\mathrm{ref}}(x)$ across $\operatorname{supp}(P_{\mathrm{ref}})$, attained by any interior point-mass reference ($r=1$; e.g., $a=1, t=26/27$ gives $D_{\mathrm{JSD}}=\psi(26/27)\approx 0.8847$).\end{proof}

\begin{proof}[Proof of Lemma 1]

On each interval of width $w$ in main-manuscript Eq. (2), at most two tent functions $w_k(c)=\max(0,1-|c/w-k|)$ are nonzero with slopes $\pm 1/w$, giving $\sum_{k=0}^6|w_k(c)-w_k(c')|\le(2/w)|c-c'|$. Telescoping across $d\in\{R,G,B\}$ using unit-sum weights on the remaining coordinates yields $\|W(\mathbf c)-W(\mathbf c')\|_1\le(2/w)\|\mathbf c-\mathbf c'\|_1\le(6/w)\|\mathbf c-\mathbf c'\|_\infty$. Averaging over paired samples displaced by at most $a$ per channel gives $\|P_{\mathrm{soft}}(S)-P_{\mathrm{soft}}(S')\|_1\le 6a/w$, so $D_{\mathrm{JSD}}(P\|Q)\le\tfrac12\|P-Q\|_1$ \cite{ref25} and $D_{\mathrm{JSD}}\le 1$ give $D_{\mathrm{JSD}}(P_{\mathrm{soft}}(S)\|P_{\mathrm{soft}}(S'))\le b$ with $b=\min(1,3a/w)$. Since $\sqrt{D_{\mathrm{JSD}}}$ is a metric \cite{ref34}, setting $x=\sqrt{\Dsvox{}(P_{\mathrm{ref}},S)}\in[0,1]$ and $y=\sqrt{\Dsvox{}(P_{\mathrm{ref}},S')}\in[0,1]$ yields $|y-x|\le\sqrt b$, and hence $|y^2-x^2|=|y-x|(y+x)\le\min(1,2\sqrt b)$ for raw $\Dsvox{}$.\end{proof}

\begin{proof}[Proof of Proposition 2]

Let $d=d_{UV}\ge 0$, and let $h=\tfrac12(d_{UU}+d_{VV})$ for V-statistics ($n,m\ge 1$, diagonals included) or $h=\tfrac12(\tilde d_{UU}+\tilde d_{VV})$ with $\tilde d_{UU}=\frac{n}{n-1}d_{UU}$, $\tilde d_{VV}=\frac{m}{m-1}d_{VV}$ for U-statistics ($n,m\ge 2$, diagonals excluded \cite{ref27}). Averaging $\|u_i-u_{i'}\|_2\le\|u_i-v_j\|_2+\|u_{i'}-v_j\|_2$ over $j\in\{1,\ldots,m\}$ and either all $(i,i')$ or off-diagonal $i\ne i'$ gives $0\le d_{UU},\tilde d_{UU}\le 2d_{UV}$, and symmetrically $0\le d_{VV},\tilde d_{VV}\le 2d_{UV}$; hence $(d,h)$ lies in the convex cone $\mathcal C=\{(d,h)\in\mathbb R^2:d\ge 0,\;0\le h\le 2d\}$. For $\delta_0>0$, $F(d,h)=(d-h)/(d+\delta_0)$ is smooth on $\mathcal C$ with


\[
\left|\frac{\partial F}{\partial d}\right|=\frac{h+\delta_0}{(d+\delta_0)^2}\le\frac{2d+\delta_0}{(d+\delta_0)^2}=\frac{(d+\delta_0)^2-d^2}{\delta_0(d+\delta_0)^2}\le\frac{1}{\delta_0},\qquad\left|\frac{\partial F}{\partial h}\right|=\frac{1}{d+\delta_0}\le\frac{1}{\delta_0}.
\]

When each point of $V$ moves by at most $\epsilon$ in Euclidean norm ($\|v'_j-v_j\|_2\le\epsilon$), every cross-pair distance changes by at most $\epsilon$ and every within-$V$ pair distance changes by at most $2\epsilon$ ($0$ on diagonals) while within-$U$ distances stay fixed, so $|d'-d|\le\epsilon$ and $|h'-h|\le\epsilon$. By the mean value theorem along the segment in $\mathcal C$ between $(d,h)$ and $(d',h')$, $|F(d',h')-F(d,h)|\le(|d'-d|+|h'-h|)/\delta_0\le 2\epsilon/\delta_0$, which simultaneously covers $\Dereg{}$ (since $\widehat{\mathcal E}_V^2\ge 0$ \cite{ref26}) and the signed U ratio. Because $z\mapsto[z]_+=\max(0,z)$ is $1$-Lipschitz and $d+\delta_0>0$, the clipped U ratio $[F(d,h)]_+=[\widehat{\mathcal E}_U^2]_+/(2d_{UV}+2\delta_0)$ satisfies $|[F(d',h')]_+-[F(d,h)]_+|\le|F(d',h')-F(d,h)|\le 2\epsilon/\delta_0$; moreover $h\ge 0$ and $\delta_0>0$ imply $0\le[F(d,h)]_+\le d/(d+\delta_0)<1$, so the clipped ratio is strictly less than $1$. Chaining $(U,V)\to(U,V')\to(U',V')$ yields $4\epsilon/\delta_0$.\end{proof}

\subsection*{Gaussian Bures--Wasserstein Baseline}

As a moment-based control, we fit sample means $\mu_U,\mu_V$ and biased sample covariances $\Sigma_U,\Sigma_V$ (divisor $|U|$ and $|V|$, zero for singletons) plus ridge $\gamma I$ ($\gamma=10^{-3}$ in squared RGB100 units), combining mean shift and Bures--Wasserstein covariance distance \cite{ref28} via


\[
\begin{aligned} &W_2^2=\|\mu_U-\mu_V\|_2^2+\operatorname{tr}\!\left(\Sigma_U+\Sigma_V-2(\Sigma_U^{1/2}\Sigma_V\Sigma_U^{1/2})^{1/2}\right), \\ &\DWtwo{}=1-e^{-W_2/50},\end{aligned}
\]

where $50$ has RGB100 length units; on Dataset B and Dataset C's four-sample $P_2$ halves, $\DWtwo{}$ yields 2.22\% and 16.67\% linear EER, respectively.
